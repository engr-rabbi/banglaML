ভূমিকা

মেশিন লার্নিং প্রকল্প শুধু একটি মডেল ট্রেন করা নয় — এটি একটি চক্রাকার (cyclical) প্রক্রিয়া যেখানে ব্যবসায়িক সমস্যা বোঝা, ডেটা সংগ্রহ, পরিষ্কারকরণ, ফিচার তৈরি, মডেল প্রশিক্ষণ, মূল্যায়ন, ডিপ্লয়মেন্ট এবং ক্রমাগত পর্যবেক্ষণ জড়িত। নিচের ইন্টারঅ্যাকটিভ প্রবাহচিত্র (উপরে) পুরো জীবনচক্রটি দেখায়। এই হ্যান্ডবুকে প্রতিটি ধাপের জন্য ব্যবহারিক (practical), কপি-পেস্ট-রেডি কোড দেওয়া হয়েছে যা সরাসরি বাস্তব প্রকল্পে প্রয়োগ করা যায়।

এনভায়রনমেন্ট সেটআপ (Conda)

যেকোনো ML প্রকল্প শুরুর আগে একটি আইসোলেটেড (isolated) Python এনভায়রনমেন্ট তৈরি করা অত্যন্ত গুরুত্বপূর্ণ, যাতে বিভিন্ন প্রকল্পের লাইব্রেরি ভার্সন একে অপরের সাথে সংঘর্ষ (conflict) না করে। এই অধ্যায়ে Conda দিয়ে এনভায়রনমেন্ট তৈরি, সক্রিয়করণ, এক্সপোর্ট ও ম্যানেজমেন্টের সম্পূর্ণ কমান্ড রেফারেন্স দেওয়া হলো।

def analyze_missing_data(dataframe):
    """
    প্রদত্ত DataFrame-এ মিসিং ডেটা বিশ্লেষণ করে।

    Parameters
    ----------
    dataframe : pd.DataFrame
        ইনপুট ডেটাফ্রেম।

    Returns
    -------
    pd.DataFrame
        Missing Percent, DataType ও Null Count সহ একটি সামারি টেবিল,
        Missing Percent অনুযায়ী descending sort করা।
    """
    # প্রতিটি কলামের মিসিং শতাংশ হিসাব
    missing_percent = (dataframe.isnull().sum() / dataframe.shape[0] * 100).sort_values(ascending=False)
    missing_percent = missing_percent.apply(lambda x: f"{x:.2f}%")

    # ডেটাটাইপ ও নাল কাউন্ট সংগ্রহ
    info_df = pd.DataFrame({
        'DataType': dataframe.dtypes,
        'Null Count': dataframe.isnull().sum()
    })

    # একত্রিত করা
    combined_df = pd.concat([missing_percent, info_df], axis=1)
    combined_df.columns = ['Missing Percent', 'DataType', 'Null Count']

    print(f'Number of rows: {dataframe.shape[0]}')
    return combined_df.sort_values(by='Missing Percent', ascending=False)

        ax[1].set_title(f'Missing Values in {name} Dataset')

def get_missing_data_indices(*datasets, names=None, top_n=100, min_nans=1, max_nans=None):
    """
    মিসিং ভ্যালুর সংখ্যা অনুযায়ী সাজানো রো-ইনডেক্সের তালিকা রিটার্ন করে।

    Parameters
    ----------
    top_n : int — সর্বোচ্চ কতগুলো রো রিটার্ন করবে (ডিফল্ট 100)
    min_nans : int — একটি রো-তে ন্যূনতম কতগুলো NaN থাকতে হবে
    max_nans : int, optional — সর্বোচ্চ কতগুলো NaN থাকতে পারবে
    """
    missing_data_info = []
    for df in datasets:
        nan_counts = df.isna().sum(axis=1)
        if max_nans is not None:
            filtered_rows = nan_counts[(nan_counts >= min_nans) & (nan_counts <= max_nans)]
        else:
            filtered_rows = nan_counts[nan_counts >= min_nans]
        sorted_row_indices = filtered_rows.sort_values(ascending=False).head(top_n).index.tolist()
        missing_data_info.extend(sorted_row_indices)
    return missing_data_info

def analyze_correlations(df, target_column="SalePrice", corr_threshold=0.7):
    """
    টার্গেট কলামের সাথে হাইলি-কোরিলেটেড ফিচারের হিটম্যাপ এবং

    Parameters
    ----------
    target_column : str — টার্গেট কলামের নাম
    corr_threshold : float — মেজর পেয়ারের জন্য থ্রেশহোল্ড (ডিফল্ট 0.7)
    """
    if target_column not in df.columns:
        raise ValueError(f"Target column '{target_column}' not found in DataFrame.")

    numeric_df = df.select_dtypes(include=['number'])
    correlation_matrix = numeric_df.corr()

    # আউটপুট ১: টার্গেটের সাথে |corr| > 0.5 ফিচারের হিটম্যাপ
    top_corr = correlation_matrix.index[abs(correlation_matrix[target_column]) > 0.5]
    if len(top_corr) > 0:
        plt.figure(figsize=(10, 8))
        sns.heatmap(numeric_df[top_corr].corr(), annot=True, cmap="YlGnBu",
                    square=True, cbar_kws={'label': 'Correlation'})
        plt.suptitle(f"Top Correlated Feature HeatMap (Correlation > 0.5 with {target_column})")
        plt.tight_layout()
        plt.show()
    else:
        print(f"No features with correlation > 0.5 with '{target_column}' found.")

    # আউটপুট ২: থ্রেশহোল্ডের উপরে মেজর কোরিলেশন পেয়ার
    abs_corr_matrix = correlation_matrix.abs()
    corr_pairs = abs_corr_matrix.stack().reset_index()
    corr_pairs.columns = ['Variable1', 'Variable2', 'Correlation']
    corr_pairs = corr_pairs[corr_pairs['Variable1'] != corr_pairs['Variable2']]

    major_corr_pairs = corr_pairs[corr_pairs['Correlation'] >= corr_threshold].reset_index(drop=True)
    major_corr_pairs = major_corr_pairs.sort_values(by=['Variable1', 'Variable2'])
    # একই পেয়ার দুইবার (A-B এবং B-A) এড়াতে

        major_corr_pairs['Variable1'] < major_corr_pairs['Variable2']
    ].reset_index(drop=True)

    print(f"\nMajor Correlation Pairs (Correlation >= {corr_threshold}):")
    print(major_corr_pairs if not major_corr_pairs.empty else f"No pairs with correlation >= {corr_threshold} found.")

def visualize_feature(df, feature, plot_type="box", target_column="SalePrice"):

    action : 'report' (শুধু রিপোর্ট), 'remove' (রো বাদ), 'cap' (শুধু iqr-এ সম্ভব)

    Returns
    -------
    pd.DataFrame — action='remove'/'cap' হলে পরিবর্তিত ডেটাফ্রেম, নাহলে মূল ডেটাফ্রেম
    """
    df_copy = df.copy()
    if column not in df_copy.columns:
        raise ValueError(f"Column '{column}' not found in DataFrame.")

    if method == 'zscore':
        z_scores = np.abs(stats.zscore(df_copy[column].dropna()))
        outliers = z_scores > threshold
        outlier_indices = df_copy[column].dropna().index[outliers]

    elif method == 'iqr':
        Q1 = df_copy[column].quantile(0.25)
        Q3 = df_copy[column].quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - threshold * IQR
        upper_bound = Q3 + threshold * IQR
        outliers = (df_copy[column] < lower_bound) | (df_copy[column] > upper_bound)
        outlier_indices = df_copy[outliers].index

    else:
        raise ValueError(f"Invalid method: '{method}'. Choose 'zscore' or 'iqr'.")

    print(f"Outliers in {column} ({method}): {len(outlier_indices)} rows")
    if len(outlier_indices) > 0:
        print(df_copy.loc[outlier_indices, [column]].head())

    if action == 'remove':
        df_copy = df_copy.drop(outlier_indices)
    elif action == 'cap' and method == 'iqr':
        df_copy[column] = df_copy[column].clip(lower=lower_bound, upper=upper_bound)
    elif action == 'cap' and method == 'zscore':
        print("Capping not supported for Z-score; use 'iqr' method or 'remove'.")

    return df_copy

def woe_encoding(df, feature, target):
    grouped = df.groupby(feature)[target].agg(['count', 'sum'])
    grouped['event_rate'] = grouped['sum'] / grouped['sum'].sum()
    grouped['non_event_rate'] = (grouped['count'] - grouped['sum']) / (grouped['count'].sum() - grouped['sum'].sum())
    grouped['woe'] = np.log(grouped['event_rate'] / grouped['non_event_rate'])
    return df[feature].map(grouped['woe'])

from sklearn.preprocessing import PolynomialFeatures
import numpy as np

def create_features(df, columns, feature_type='poly', degree=2):
    """
    DataFrame-এ নতুন ফিচার তৈরি করে (polynomial, interaction, অথবা log transformation)।

    feature_type: 'poly', 'interaction', অথবা 'log'
    """
    df_copy = df.copy()

    if feature_type == 'poly':
        poly = PolynomialFeatures(degree=degree, include_bias=False)
        poly_features = poly.fit_transform(df_copy[columns])
        poly_columns = poly.get_feature_names_out(columns)
        df_poly = pd.DataFrame(poly_features, columns=poly_columns, index=df_copy.index)
        df_copy = pd.concat([df_copy, df_poly], axis=1)

    elif feature_type == 'interaction':
        for i, col1 in enumerate(columns):
            for col2 in columns[i+1:]:
                df_copy[f"{col1}_{col2}_interaction"] = df_copy[col1] * df_copy[col2]

    elif feature_type == 'log':
        for col in columns:
            if (df_copy[col] <= 0).any():
                print(f"Skipping log for {col}: Contains non-positive values.")
                continue
            df_copy[f"{col}_log"] = np.log(df_copy[col])

    else:
        raise ValueError(f"Invalid feature_type: '{feature_type}'. Choose 'poly', 'interaction', or 'log'.")

    return df_copy

৪. TF-IDF (Term Frequency – Inverse Document Frequency)

ধারণা: TF হলো ডকুমেন্টে শব্দের ফ্রিকোয়েন্সি, IDF হলো log(N / (1+df)) — যেখানে N মোট ডকুমেন্ট সংখ্যা, df শব্দের ডকুমেন্ট ফ্রিকোয়েন্সি। এটি রেয়ার শব্দকে বেশি ওয়েট দেয়। সূত্র: TF-IDF = TF × log(N / (1 + df))।

কখন ব্যবহার করবেন: যখন কমন শব্দ ইগনোর করতে চান, যেমন সার্চ ইঞ্জিনে। ডকুমেন্ট লম্বা হলে বা কমন শব্দ বেশি থাকলে সাধারণত TF-IDF, BoW-এর চেয়ে ভালো ফলাফল দেয়।

    ('imputer', SimpleImputer(strategy='median')),
    ('scaler', StandardScaler())
])

    ('imputer', SimpleImputer(strategy='constant', fill_value='missing')),
    ('onehot', OneHotEncoder(handle_unknown='ignore'))
])

    ('num', numeric_transformer, numeric_features),
    ('cat', categorical_transformer, categorical_features)
])

    ('preprocessor', preprocessor),

def detect_drift(reference_df, current_df, threshold=0.05):
    """
    reference_df (ট্রেনিং ডেটা) এবং current_df (নতুন ডেটা)-এর মধ্যে
    একটি সাধারণ পরিসংখ্যানিক ড্রিফট চেক করে।
    """
    drift_report = {}
    for col in reference_df.columns:
        if reference_df[col].dtype == 'object':
            # ক্যাটেগরিক্যাল: কমন ক্যাটাগরির অনুপাতের পার্থক্য যোগফল
            ref_dist = reference_df[col].value_counts(normalize=True)
            cur_dist = current_df[col].value_counts(normalize=True)
            common_cats = set(ref_dist.index) & set(cur_dist.index)
            diff = sum(abs(ref_dist[cat] - cur_dist[cat]) for cat in common_cats)
            if diff > threshold:
                drift_report[col] = f"Drift detected (diff: {diff:.3f})"
        else:
            # নিউমেরিক: গড়ের আপেক্ষিক পার্থক্য
            if abs(reference_df[col].mean() - current_df[col].mean()) / reference_df[col].mean() > threshold:
                drift_report[col] = "Mean drift"
    return drift_report

import pandas as pd

def query_dataframe(df, query_type, **kwargs):
    """
    একটি pandas DataFrame-এ বিভিন্ন ধরনের কুয়েরি/অপারেশন সম্পাদন করে।

    Parameters
    ----------
    query_type : 'filter', 'group', 'sort', 'stats'
    **kwargs :
        filter → condition (str, যেমন 'OpenPorchSF > 400')
        group  → group_by (str/list), agg (dict/str, যেমন {'SalePrice': 'mean'})
        sort   → sort_by (str/list), ascending (bool, ডিফল্ট True)
        stats  → column (str, ঐচ্ছিক — না দিলে সব নিউমেরিক কলামের সামারি)

    Returns
    -------
    filter → ফিল্টার করা DataFrame
    group  → গ্রুপড ও অ্যাগ্রিগেটেড DataFrame
    sort   → সর্টেড DataFrame
    stats  → পরিসংখ্যানিক সামারি প্রিন্ট করে (রিটার্ন ভ্যালু নেই)
    """
    if not isinstance(df, pd.DataFrame):
        raise ValueError("Input 'df' must be a pandas DataFrame.")

    if query_type.lower() == "filter":
        condition = kwargs.get("condition")
        if not condition:
            raise ValueError("For 'filter', provide a 'condition' string (e.g., 'OpenPorchSF > 400').")
        return df.query(condition)

    elif query_type.lower() == "group":
        group_by = kwargs.get("group_by")
        agg = kwargs.get("agg", "mean")
        if not group_by:
            raise ValueError("For 'group', provide 'group_by' column(s).")
        return df.groupby(group_by).agg(agg).reset_index()

    elif query_type.lower() == "sort":
        sort_by = kwargs.get("sort_by")
        ascending = kwargs.get("ascending", True)
        if not sort_by:
            raise ValueError("For 'sort', provide 'sort_by' column(s).")
        return df.sort_values(by=sort_by, ascending=ascending)

    elif query_type.lower() == "stats":
        column = kwargs.get("column")
        if column:
            if column not in df.columns:
                raise ValueError(f"Column '{column}' not found in DataFrame.")
            if pd.api.types.is_numeric_dtype(df[column]):
                print(f"Statistical Summary for {column}:")
                print(df[column].describe())
                print(f"\nSkewness: {df[column].skew():.3f}")
                print(f"Kurtosis: {df[column].kurtosis():.3f}")
                print(f"Missing Values: {df[column].isna().sum()}")
            else:

def vlookup(df, lookup_df, key_column, lookup_column, return_column, how="left"):
    """
    একটি lookup DataFrame-এর সাথে merge করে VLOOKUP-এর মতো অপারেশন করে।

    Parameters
    ----------
    df : pandas DataFrame — মূল ডেটাফ্রেম
    lookup_df : pandas DataFrame — যেখান থেকে মান আনা হবে
    key_column : str — df-এর যে কলাম মিলাতে হবে
    lookup_column : str — lookup_df-এর যে কলাম মিলাতে হবে
    return_column : str — lookup_df থেকে যে কলামের মান আনতে হবে
    how : 'left', 'inner', 'right', 'outer' (ডিফল্ট 'left')

    Returns
    -------
    df-এর সব কলাম + return_column সহ একটি নতুন DataFrame
    """
    if not isinstance(df, pd.DataFrame) or not isinstance(lookup_df, pd.DataFrame):
        raise ValueError("Both 'df' and 'lookup_df' must be pandas DataFrames.")
    if key_column not in df.columns:
        raise ValueError(f"Key column '{key_column}' not found in main DataFrame.")
    if lookup_column not in lookup_df.columns:
        raise ValueError(f"Lookup column '{lookup_column}' not found in lookup DataFrame.")
    if return_column not in lookup_df.columns:
        raise ValueError(f"Return column '{return_column}' not found in lookup DataFrame.")
    if how not in ['left', 'inner', 'right', 'outer']:
        raise ValueError(f"Invalid 'how' parameter: '{how}'.")

    try:
        result = df.merge(
            lookup_df[[lookup_column, return_column]],
            left_on=key_column, right_on=lookup_column, how=how
        )
        if lookup_column != key_column:
            result = result.drop(columns=lookup_column, errors='ignore')
        return result
    except Exception as e:
        raise ValueError(f"VLOOKUP operation failed: {e}")

ডেটাসেট ছোট থেকে মাঝারি হলে এবং ফিচারগুলোর মধ্যে সরল সম্পর্ক থাকলে KNN Imputer দ্রুত ও কার্যকর। জটিল, বহু-স্তরীয় আন্তঃফিচার সম্পর্ক থাকলে এবং প্রতিটি মিসিং ভ্যালুর জন্য একাধিক সম্ভাব্য মান বিবেচনা করতে চাইলে (uncertainty quantification) MICE বেশি উপযুক্ত, যদিও এটি কম্পিউটেশনালি ব্যয়বহুল।

GridSearchCV অনেক সময় নিচ্ছে — কী করব?

১. ট্রান্সফরমেশন পুরো ডেটাসেটে ফিট করা

স্কেলার/এনকোডার/ইম্পিউটার train+test একসাথে fit() করলে টেস্ট সেটের তথ্য ট্রেনিং-এ লিক হয়ে যায়। সবসময় fit() শুধু ট্রেনিং ডেটায় করুন, তারপর transform() উভয় সেটে প্রয়োগ করুন — অথবা Pipeline ব্যবহার করুন যা এটি স্বয়ংক্রিয়ভাবে নিশ্চিত করে।

২. স্প্লিটের আগে SMOTE/রি-স্যাম্পলিং প্রয়োগ করা

পুরো ডেটাসেটে SMOTE প্রয়োগ করে তারপর ট্রেন-টেস্ট স্প্লিট করলে সিন্থেটিক ডেটা টেস্ট সেটে লিক হয়ে যেতে পারে। স্প্লিট করার পর শুধুমাত্র ট্রেনিং সেটে SMOTE প্রয়োগ করুন।

৩. টাইম-সিরিজ ডেটা র‍্যান্ডম শাফল করে স্প্লিট করা

এতে ভবিষ্যতের তথ্য ট্রেনিং সেটে ঢুকে যায়, যা বাস্তব-বিশ্বে অসম্ভব একটি পরিস্থিতি তৈরি করে এবং মডেলের ফলাফল বিভ্রান্তিকরভাবে ভালো দেখায়।

৪. ইমব্যালেন্সড ডেটায় শুধু Accuracy দেখা

৯৯% ক্লাস A এবং ১% ক্লাস B হলে, সবসময় "A" প্রেডিক্ট করা মডেলও ৯৯% accuracy পাবে অথচ এটি সম্পূর্ণ অকার্যকর। Precision, Recall, F1-score বা AUC-ROC-এর দিকে নজর দিন।

৫. উচ্চ-কার্ডিনালিটি ফিচারে One-Hot Encoding ব্যবহার

শত শত ইউনিক ক্যাটাগরি থাকলে One-Hot Encoding প্রচুর কলাম তৈরি করে (dimensionality explosion), যা মেমরি সমস্যা ও ওভারফিটিং ঘটাতে পারে — এক্ষেত্রে Target/Frequency/Hashing Encoding বিবেচনা করুন।

৬. random_state সেট না করা

এর ফলে প্রতিবার কোড রান করলে ভিন্ন ফলাফল আসতে পারে, যা ডিবাগিং ও রেজাল্ট তুলনা কঠিন করে তোলে — সবসময় একটি ফিক্সড সিড ব্যবহার করুন।

এই হ্যান্ডবুকে আমরা একটি সম্পূর্ণ মেশিন লার্নিং প্রকল্পের জীবনচক্র — এনভায়রনমেন্ট সেটআপ থেকে শুরু করে ডেটা এক্সপ্লোরেশন, প্রি-প্রসেসিং, ফিচার ইঞ্জিনিয়ারিং, মডেল ট্রেনিং, টিউনিং, ইভালুয়েশন, ব্যাখ্যাযোগ্যতা এবং সবশেষে ডিপ লার্নিং পর্যন্ত — একত্রিত ও সুসংগঠিতভাবে কভার করেছি। প্রতিটি অধ্যায়ে ব্যবহারিক, কপি-পেস্ট-রেডি কোড এবং কখন কোন পদ্ধতি বেছে নেবেন তার স্পষ্ট নির্দেশনা দেওয়া হয়েছে।

মনে রাখা জরুরি যে মেশিন লার্নিং একটি চক্রাকার (iterative) প্রক্রিয়া — প্রথম মডেলটিই চূড়ান্ত হবে না। ডেটা বোঝা, পরীক্ষা-নিরীক্ষা করা এবং ধারাবাহিকভাবে মনিটর ও উন্নত করার মানসিকতা নিয়ে এগিয়ে যাওয়াই দীর্ঘমেয়াদে সবচেয়ে ভালো ফলাফল দেয়।
